cmake_minimum_required(VERSION 3.16.3)
project(OpenHDOdid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ODID_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(ODID_CORE "${ODID_REPO}/third_party/opendroneid-core-c/libopendroneid")
if(NOT EXISTS "${ODID_CORE}/opendroneid.c")
  message(FATAL_ERROR "Initialize pinned third_party/opendroneid-core-c submodule")
endif()
find_package(Threads REQUIRED)
add_library(openhd_odid_core STATIC Core.cpp ConfigJson.cpp "${ODID_CORE}/opendroneid.c")
target_include_directories(openhd_odid_core PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}" "${ODID_CORE}"
  "${ODID_REPO}/OpenHD/ohd_common/inc"
  "${ODID_REPO}/OpenHD/ohd_telemetry/lib/mavlink-headers/mavlink/v2.0"
  "${ODID_REPO}/OpenHD/ohd_common/lib/json/include")
target_compile_options(openhd_odid_core PUBLIC -Wno-address-of-packed-member -Wno-psabi)
target_link_libraries(openhd_odid_core PUBLIC Threads::Threads m)
option(OPENHD_ODID_BLUEZ "Build experimental opt-in BlueZ Legacy advertising" OFF)
if(OPENHD_ODID_BLUEZ)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ODID_GIO REQUIRED IMPORTED_TARGET gio-2.0)
  target_sources(openhd_odid_core PRIVATE BluezBackend.cpp)
  target_link_libraries(openhd_odid_core PRIVATE PkgConfig::ODID_GIO)
  target_compile_definitions(openhd_odid_core PRIVATE OPENHD_ODID_BLUEZ=1)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(openhd_odid_test test.cpp Integration.cpp)
  target_link_libraries(openhd_odid_test PRIVATE openhd_odid_core)
  add_test(NAME openhd_odid_test COMMAND openhd_odid_test)
endif()
